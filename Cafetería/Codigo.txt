9. CAFETERÍA DE ESPECIALIDAD

En el módulo de ventas de una cafetería (clase mayor: MCafeteria) se registran las bebidas preparadas (clase menor base: MBebida), conociendo su nombre y precio base. Existen dos preparaciones especializadas: Café Caliente (MCafeCaliente, que suma $0.75 al precio base si se solicita con leche vegetal) y Bebida Fría / Frappé (MBebidaFria, que incrementa el precio base en un 20% si es tamaño grande).

Se desea calcular el precio final de cada bebida, el monto total de ventas y cuántas bebidas superaron los $4 de precio final.

Nuevo Requerimiento:
Se solicita implementar el método polimórfico gramosAzucar() (contenido estimado en gramos) en la jerarquía de bebidas:
- Clase base (MBebida): aporta 10 gramos base de endulzante o jarabe estándar.
- Café Caliente (MCafeCaliente): reduce a la mitad los gramos base si se prepara con leche vegetal; de lo contrario mantiene los gramos base.
- Bebida Fría / Frappé (MBebidaFria): suma 12 gramos adicionales a los gramos base por la mezcla fría; además, si es tamaño grande, incrementa ese subtotal en un 20%.
Actualización en MCafeteria: Reportar los gramos de azúcar de cada bebida y el promedio de gramos de azúcar por bebida vendida.

Datos de entrada:
- Bebida: Latte, Precio base: 3, Tipo: Caliente, Leche vegetal: Sí
- Bebida: Frappé Mocha, Precio base: 4, Tipo: Fría, Grande: Sí
- Bebida: Americano, Precio base: 2.50, Tipo: Caliente, Leche vegetal: No
- Bebida: Matcha Helado, Precio base: 4.50, Tipo: Fría, Grande: No

La salida requerida presenta el siguiente formato:
Latte: $3.75
Frappé Mocha: $4.80
Americano: $2.50
Matcha Helado: $4.50

Monto total de ventas: $15.55
Bebidas con precio mayor a $4: 2

Salida esperada del nuevo requerimiento:
Latte -> Azúcar: 5.00 g
Frappé Mocha -> Azúcar: 26.40 g
Americano -> Azúcar: 10.00 g
Matcha Helado -> Azúcar: 22.00 g

Promedio de azúcar por bebida: 15.85 g

import { MBebida } from "./MBebida.js";

export class MCafeCaliente extends MBebida {
  private static readonly RECARGO_LECHE_VEGETAL: number = 0.75;
  private static readonly FACTOR_AZUCAR_LECHE_VEGETAL: number = 0.5;
  private _conLecheVegetal: boolean = false;

  constructor(nombre: string, precioBase: number, conLecheVegetal: boolean) {
    super(nombre, precioBase);
    this.conLecheVegetal = conLecheVegetal;
  }

  set conLecheVegetal(valor: boolean) {
    this._conLecheVegetal = Boolean(valor);
  }

  get conLecheVegetal(): boolean {
    return this._conLecheVegetal;
  }

  override calcularPrecioFinal(): number {
    let precio = super.calcularPrecioFinal();
    if (this.conLecheVegetal) {
      precio += MCafeCaliente.RECARGO_LECHE_VEGETAL;
    }
    return precio;
  }

  override gramosAzucar(): number {
    const gramosBase = super.gramosAzucar();
    if (this.conLecheVegetal) {
      return gramosBase * MCafeCaliente.FACTOR_AZUCAR_LECHE_VEGETAL;
    }
    return gramosBase;
  }

  override obtenerTipo(): string {
    return "Café Caliente";
  }
}

export class MBebidaFria extends MBebida {
  private static readonly INCREMENTO_TAMANO_GRANDE: number = 0.20;
  private static readonly AZUCAR_MEZCLA_FRIA: number = 12;
  private _esGrande: boolean = false;

  constructor(nombre: string, precioBase: number, esGrande: boolean) {
    super(nombre, precioBase);
    this.esGrande = esGrande;
  }

  set esGrande(valor: boolean) {
    this._esGrande = Boolean(valor);
  }

  get esGrande(): boolean {
    return this._esGrande;
  }

  override calcularPrecioFinal(): number {
    let precio = super.calcularPrecioFinal();
    if (this.esGrande) {
      precio += precio * MBebidaFria.INCREMENTO_TAMANO_GRANDE;
    }
    return precio;
  }

  override gramosAzucar(): number {
    let subtotal = super.gramosAzucar() + MBebidaFria.AZUCAR_MEZCLA_FRIA;
    if (this.esGrande) {
      subtotal += subtotal * MBebidaFria.INCREMENTO_TAMANO_GRANDE;
    }
    return subtotal;
  }

  override obtenerTipo(): string {
    return "Bebida Fría";
  }
}

import { MBebida } from "./MBebida.js";

export class MCafeteria {
  private static readonly PRECIO_LIMITE: number = 4;
  private acMontoTotal: number = 0;
  private acGramosAzucar: number = 0;
  private cntBebidasMayorCuatro: number = 0;
  private cntBebidas: number = 0;

  procesarBebida(bebida: MBebida): void {
    const precioFinal = bebida.calcularPrecioFinal();
    this.acMontoTotal += precioFinal;
    this.acGramosAzucar += bebida.gramosAzucar();
    this.cntBebidas++;

    if (precioFinal > MCafeteria.PRECIO_LIMITE) {
      this.cntBebidasMayorCuatro++;
    }
  }

  montoTotalVentas(): number {
    return this.acMontoTotal;
  }

  cantidadBebidasMayorCuatro(): number {
    return this.cntBebidasMayorCuatro;
  }

  promedioGramosAzucar(): number {
    if (this.cntBebidas === 0) {
      return 0;
    }
    return this.acGramosAzucar / this.cntBebidas;
  }

  cantidadBebidas(): number {
    return this.cntBebidas;
  }

  reiniciar(): void {
    this.acMontoTotal = 0;
    this.acGramosAzucar = 0;
    this.cntBebidasMayorCuatro = 0;
    this.cntBebidas = 0;
  }
}

import { MCafeteria } from "../models/MCafeteria.js";
import { MBebida } from "../models/MBebida.js";
import { MCafeCaliente } from "../models/MCafeCaliente.js";
import { MBebidaFria } from "../models/MBebidaFria.js";
import { ICafeteriaView } from "../interfaces/ICafeteriaView.js";

export class cCafeteria {
  private modelo: MCafeteria;
  private vista: ICafeteriaView;

  constructor(modeloIny: MCafeteria, vistaIny: ICafeteriaView) {
    this.modelo = modeloIny;
    this.vista = vistaIny;
  }

  iniciar(): void {
    this.vista.onProcesar(() => this.procesarBebida());
    this.vista.onCargarCasosPrueba(() => this.cargarCasosPrueba());
    this.vista.onLimpiar(() => this.reiniciar());
    this.cargarCasosPrueba();
  }

  private registrarBebida(bebida: MBebida): void {
    this.modelo.procesarBebida(bebida);
    this.vista.reportarBebida(
      bebida.nombre,
      bebida.obtenerTipo(),
      bebida.precioBase,
      bebida.calcularPrecioFinal(),
      bebida.gramosAzucar()
    );
  }

  private actualizarResumen(): void {
    this.vista.reportarResumen(
      this.modelo.montoTotalVentas(),
      this.modelo.cantidadBebidasMayorCuatro(),
      this.modelo.promedioGramosAzucar(),
      this.modelo.cantidadBebidas()
    );
  }

  private procesarBebida(): void {
    const nombre = this.vista.nombre;
    const precioBase = this.vista.precioBase;
    const tipo = this.vista.tipo;

    if (!nombre) {
      this.vista.mostrarMensaje("Por favor, ingrese el nombre de la bebida.", "warning");
      return;
    }

    if (isNaN(precioBase) || precioBase <= 0) {
      this.vista.mostrarMensaje("El precio base debe ser un número positivo mayor a 0.", "danger");
      return;
    }

    let bebida: MBebida;

    if (tipo === "caliente") {
      bebida = new MCafeCaliente(nombre, precioBase, this.vista.conLecheVegetal);
    } else {
      bebida = new MBebidaFria(nombre, precioBase, this.vista.esGrande);
    }

    this.registrarBebida(bebida);
    this.actualizarResumen();
    this.vista.limpiarFormulario();
    this.vista.mostrarMensaje(`Bebida ${bebida.nombre} registrada correctamente.`, "success");
  }

  private cargarCasosPrueba(): void {
    this.modelo.reiniciar();
    this.vista.reiniciarVista();

    const bebida1 = new MCafeCaliente("Latte", 3, true);
    const bebida2 = new MBebidaFria("Frappé Mocha", 4, true);
    const bebida3 = new MCafeCaliente("Americano", 2.50, false);
    const bebida4 = new MBebidaFria("Matcha Helado", 4.50, false);

    this.registrarBebida(bebida1);
    this.registrarBebida(bebida2);
    this.registrarBebida(bebida3);
    this.registrarBebida(bebida4);

    this.actualizarResumen();
    this.vista.mostrarMensaje("Casos de prueba cargados correctamente.", "info");
  }

  private reiniciar(): void {
    this.modelo.reiniciar();
    this.vista.reiniciarVista();
    this.vista.mostrarMensaje("Ventas reiniciadas correctamente.", "info");
  }
}
